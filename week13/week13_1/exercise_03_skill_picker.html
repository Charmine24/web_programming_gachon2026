Skill Picker

Six skills as checkboxes with a live counter. Selecting more than 3 shows a warning.

Please select at most 3 skills.

React version (from the slides)

// Week 13 — Exercise 03
import { useState } from 'react';

// ─── Exercise 3: Skill picker ───────────────────────────────────────────────
// 6 skills as checkboxes, live counter, warning if more than 3 selected.
const SKILLS = ['HTML', 'CSS', 'JavaScript', 'React', 'Node.js', 'Python'];

function SkillPicker() {
  const [selected, setSelected] = useState([]);

  function toggle(skill, checked) {
    setSelected(prev => (checked ? [...prev, skill] : prev.filter(s => s !== skill)));
  }

  return (
    <div>
      <p>Selected: {selected.length} / 3</p>
      {selected.length > 3 && <p style={{ color: 'red' }}>Please select at most 3 skills.</p>}
      {SKILLS.map(skill => (
        <label key={skill}>
          <input
            type="checkbox"
            checked={selected.includes(skill)}
            onChange={e => toggle(skill, e.target.checked)}
          />
          {skill}
        </label>
      ))}
    </div>
  );
}

export default SkillPicker;
