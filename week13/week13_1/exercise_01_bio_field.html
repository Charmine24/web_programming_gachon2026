Bio Field with Character Limit

A controlled textarea with a 200-character limit and a live count. .slice(0, MAX) stops typing at the limit, and the count turns red in the last 20 characters.

React version (from the slides)

// Week 13 — Exercise 01
import { useState } from 'react';

// ─── Exercise 1: Bio field with character limit ─────────────────────────────
// Controlled textarea, max 200 chars, live "X / 200" count.
// .slice(0, MAX) stops typing once the limit is reached.
function BioField() {
  const [bio, setBio] = useState('');
  const MAX = 200;
  const remaining = MAX - bio.length;

  return (
    <div>
      <textarea
        value={bio}
        onChange={e => setBio(e.target.value.slice(0, MAX))}
        rows={4}
        placeholder="Your bio"
      />
      <p style={{ color: remaining < 20 ? 'red' : 'gray', fontSize: 13 }}>
        {bio.length} / {MAX}
      </p>
    </div>
  );
}

export default BioField;
