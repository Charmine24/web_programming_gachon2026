Form Validation

Validation runs on submit, errors are stored in an object keyed by field name, and each error clears as soon as you type in that field. Try submitting empty fields, a bad email or a short password.

Account created!

React version (from the slides)

// Week 13 — Topic 05
import { useState } from 'react';

const errorStyle = { color: 'red', fontSize: 13 };

// ─── 6. Form Validation ─────────────────────────────────────────────────────
// Validate on submit, store errors keyed by field name, clear an error as the user types.
function SignupForm() {
  const [form, setForm] = useState({ email: '', password: '' });
  const [errors, setErrors] = useState({});

  function validate() {
    const e = {};
    if (!form.email) {
      e.email = 'Email is required';
    } else if (!/\S+@\S+\.\S+/.test(form.email)) {
      e.email = 'Enter a valid email address';
    }
    if (!form.password) {
      e.password = 'Password is required';
    } else if (form.password.length < 8) {
      e.password = 'Password must be at least 8 characters';
    }
    return e;
  }

  function handleChange(e) {
    const { name, value } = e.target;
    setForm(prev => ({ ...prev, [name]: value }));
    if (errors[name]) setErrors(prev => ({ ...prev, [name]: '' }));
  }

  function handleSubmit(e) {
    e.preventDefault();
    const newErrors = validate();
    if (Object.keys(newErrors).length > 0) {
      setErrors(newErrors);
      return;
    }
    alert('Account created!');
  }

  return (
    <form onSubmit={handleSubmit}>
      <div>
        <input name="email" value={form.email} onChange={handleChange}
               type="email" placeholder="Email"
               style={{ borderColor: errors.email ? 'red' : '' }} />
        {errors.email && <p style={errorStyle}>{errors.email}</p>}
      </div>
      <div>
        <input name="password" value={form.password} onChange={handleChange}
               type="password" placeholder="Password"
               style={{ borderColor: errors.password ? 'red' : '' }} />
        {errors.password && <p style={errorStyle}>{errors.password}</p>}
      </div>
      <button type="submit">Create account</button>
    </form>
  );
}

export default SignupForm;
