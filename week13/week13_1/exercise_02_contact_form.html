Contact Form

Name, Email and Message are all required, and each error appears next to its field. A successful send shows a thank-you message and resets the form.

React version (from the slides)

// Week 13 — Exercise 02
import { useState } from 'react';

const errorStyle = { color: 'red', fontSize: 13 };

// ─── Exercise 2: Contact form ───────────────────────────────────────────────
// Name, Email, Message — all required. Errors next to each field.
// On success: "Thank you" message and form reset.
function ContactForm() {
  const EMPTY = { name: '', email: '', message: '' };
  const [form, setForm] = useState(EMPTY);
  const [errors, setErrors] = useState({});
  const [submittedName, setSubmittedName] = useState('');

  function handleChange(e) {
    const { name, value } = e.target;
    setForm(p => ({ ...p, [name]: value }));
    setErrors(p => ({ ...p, [name]: '' }));
  }

  function handleSubmit(e) {
    e.preventDefault();
    const errs = {};
    if (!form.name.trim()) errs.name = 'Name is required';
    if (!form.email.trim()) errs.email = 'Email is required';
    if (!form.message.trim()) errs.message = 'Message is required';
    if (Object.keys(errs).length) { setErrors(errs); return; }
    setSubmittedName(form.name.trim());
    setForm(EMPTY); // reset the form
  }

  if (submittedName) {
    return (
      <div>
        <p>Thank you, {submittedName}! We'll be in touch.</p>
        <button onClick={() => setSubmittedName('')}>Send another</button>
      </div>
    );
  }

  return (
    <form onSubmit={handleSubmit}>
      {['name', 'email', 'message'].map(field => (
        <div key={field}>
          <label>
            {field[0].toUpperCase() + field.slice(1)}<br />
            {field === 'message'
              ? <textarea name={field} value={form[field]} onChange={handleChange} rows={4} />
              : <input name={field} value={form[field]} onChange={handleChange} />}
          </label>
          {errors[field] && <p style={errorStyle}>{errors[field]}</p>}
        </div>
      ))}
      <button type="submit">Send</button>
    </form>
  );
}

export default ContactForm;
