Select and Textarea

A <select> and a <textarea> are controlled the same way as text inputs: value plus onChange. In React, a textarea uses the value prop, not children.

React version (from the slides)

  
// Week 13 — Topic 02
import { useState } from 'react';

// ─── 3. Select & Textarea ───────────────────────────────────────────────────
// Both use value + onChange. In React, textarea uses the value prop, not children.
function SelectAndTextarea() {
  const [department, setDepartment] = useState('');
  const [bio, setBio] = useState('');

  return (
    <div>
      <select value={department} onChange={e => setDepartment(e.target.value)}>
        <option value="">— Select department —</option>
        <option value="cs">Computer Science</option>
        <option value="ee">Electrical Engineering</option>
        <option value="me">Mechanical Engineering</option>
      </select>
      <textarea
        value={bio}
        onChange={e => setBio(e.target.value)}
        rows={4}
        placeholder="Tell us about yourself..."
      />
    </div>
  );
}

export default SelectAndTextarea;
