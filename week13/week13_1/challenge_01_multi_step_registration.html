// Week 13 — Challenge: Multi-Step Registration Wizard
//
// Step 1 – Personal Info:  Name (required, min 2 chars), Email (must match regex)
// Step 2 – Account Setup:  Username (min 3 chars, no spaces), Password (min 8),
//                          Confirm Password (must match password)
// Step 3 – Preferences:    Role (Student | TA | Instructor radios),
//                          Subscribe to newsletter (checkbox), review summary
//
// Requirements:
//  - "Next" stays disabled until the current step is valid
//  - Shows progress: "Step 2 of 3" plus a progress bar
//  - "Previous" keeps data already entered (all data lives in ONE state object)
//  - Final submit: console.log(fullFormObject)
import { useState } from 'react';

const TOTAL_STEPS = 3;
const ROLES = ['Student', 'TA', 'Instructor'];

const INITIAL_FORM = {
  name: '',
  email: '',
  username: '',
  password: '',
  confirmPassword: '',
  role: '',
  newsletter: false,
};

// Fields that belong to each step (used to validate only the current step)
const STEP_FIELDS = {
  1: ['name', 'email'],
  2: ['username', 'password', 'confirmPassword'],
  3: ['role'],
};

// Returns an error object for the whole form; empty object = everything valid
function validate(form) {
  const errs = {};

  // Step 1
  if (form.name.trim().length < 2) {
    errs.name = 'Name must be at least 2 characters';
  }
  if (!/^\S+@\S+\.\S+$/.test(form.email)) {
    errs.email = 'Enter a valid email address';
  }

  // Step 2
  if (form.username.length < 3) {
    errs.username = 'Username must be at least 3 characters';
  } else if (/\s/.test(form.username)) {
    errs.username = 'Username cannot contain spaces';
  }
  if (form.password.length < 8) {
    errs.password = 'Password must be at least 8 characters';
  }
  if (!form.confirmPassword || form.confirmPassword !== form.password) {
    errs.confirmPassword = 'Passwords do not match';
  }

  // Step 3
  if (!form.role) {
    errs.role = 'Please choose a role';
  }

  return errs;
}

// ─── Styles ─────────────────────────────────────────────────────────────────
const styles = {
  card: {
    maxWidth: 420, margin: '40px auto', padding: 24, borderRadius: 14,
    background: '#fff', boxShadow: '0 4px 20px rgba(0,0,0,0.12)',
    fontFamily: 'system-ui, sans-serif',
  },
  progress: { display: 'flex', gap: 8, margin: '12px 0 20px' },
  bar: done => ({
    flex: 1, height: 10, borderRadius: 6,
    background: done ? '#1fb5d4' : '#d9dfeb',
  }),
  input: hasError => ({
    display: 'block', width: '100%', boxSizing: 'border-box',
    padding: '12px 14px', marginTop: 12, fontSize: 15, borderRadius: 8,
    background: '#eef2fa',
    border: `2px solid ${hasError ? '#e0435b' : '#d9dfeb'}`,
  }),
  error: { color: '#e0435b', fontSize: 13, margin: '4px 0 0' },
  buttons: { display: 'flex', gap: 12, marginTop: 24 },
  btn: disabled => ({
    flex: 1, padding: '12px 0', fontSize: 15, fontWeight: 600,
    border: 'none', borderRadius: 8, color: '#fff',
    background: disabled ? '#aab4cc' : '#0d1526',
    cursor: disabled ? 'not-allowed' : 'pointer',
  }),
  summary: { background: '#f5f7fb', borderRadius: 8, padding: 12, marginTop: 16, fontSize: 14 },
};

export default function MultiStepRegistration() {
  const [step, setStep] = useState(1);
  const [form, setForm] = useState(INITIAL_FORM);
  const [touched, setTouched] = useState({}); // only show errors after the user types in a field
  const [submitted, setSubmitted] = useState(false);

  const errors = validate(form);
  const stepIsValid = STEP_FIELDS[step].every(field => !errors[field]);

  // One handler for all inputs; checkboxes use `checked`, everything else `value`
  function handleChange(e) {
    const { name, value, type, checked } = e.target;
    setForm(prev => ({ ...prev, [name]: type === 'checkbox' ? checked : value }));
    setTouched(prev => ({ ...prev, [name]: true }));
  }

  function showError(field) {
    return touched[field] && errors[field];
  }

  function next() {
    if (stepIsValid) setStep(s => s + 1);
  }

  function previous() {
    setStep(s => s - 1); // form state is untouched, so earlier data is kept
  }

  function handleSubmit(e) {
    e.preventDefault();
    if (Object.keys(errors).length > 0) return;
    console.log(form); // final submit: full form object
    setSubmitted(true);
  }

  if (submitted) {
    return (
      <div style={styles.card}>
        <h2>Registration complete 🎉</h2>
        <p>Welcome, {form.name}! Check the console for the submitted data.</p>
      </div>
    );
  }

  function field(name, placeholder, type = 'text') {
    return (
      <div>
        <input
          name={name}
          type={type}
          placeholder={placeholder}
          value={form[name]}
          onChange={handleChange}
          style={styles.input(showError(name))}
        />
        {showError(name) && <p style={styles.error}>{errors[name]}</p>}
      </div>
    );
  }

  return (
    <form style={styles.card} onSubmit={handleSubmit} noValidate>
      <h2 style={{ margin: 0 }}>Step {step} of {TOTAL_STEPS}</h2>

      {/* Progress bar */}
      <div style={styles.progress}>
        {Array.from({ length: TOTAL_STEPS }, (_, i) => (
          <div key={i} style={styles.bar(i < step)} />
        ))}
      </div>

      {/* Step 1 – Personal Info */}
      {step === 1 && (
        <>
          {field('name', 'Name')}
          {field('email', 'Email', 'email')}
        </>
      )}

      {/* Step 2 – Account Setup */}
      {step === 2 && (
        <>
          {field('username', 'Username')}
          {field('password', 'Password', 'password')}
          {field('confirmPassword', 'Confirm', 'password')}
        </>
      )}

      {/* Step 3 – Preferences + Review */}
      {step === 3 && (
        <>
          <fieldset style={{ border: 'none', padding: 0, marginTop: 12 }}>
            <legend style={{ fontWeight: 600 }}>Role</legend>
            {ROLES.map(role => (
              <label key={role} style={{ marginRight: 16 }}>
                <input
                  type="radio"
                  name="role"
                  value={role}
                  checked={form.role === role}
                  onChange={handleChange}
                />
                {' '}{role}
              </label>
            ))}
          </fieldset>

          <label style={{ display: 'block', marginTop: 12 }}>
            <input
              type="checkbox"
              name="newsletter"
              checked={form.newsletter}
              onChange={handleChange}
            />
            {' '}Subscribe to newsletter
          </label>

          {/* Read-only review summary */}
          <div style={styles.summary}>
            <strong>Review</strong>
            <p>Name: {form.name}</p>
            <p>Email: {form.email}</p>
            <p>Username: {form.username}</p>
            <p>Password: {'•'.repeat(form.password.length)}</p>
            <p>Role: {form.role || '—'}</p>
            <p>Newsletter: {form.newsletter ? 'Yes' : 'No'}</p>
          </div>
        </>
      )}

      {/* Navigation */}
      <div style={styles.buttons}>
        {step > 1 && (
          <button type="button" onClick={previous} style={styles.btn(false)}>
            Previous
          </button>
        )}
        {step < TOTAL_STEPS ? (
          <button type="button" onClick={next} disabled={!stepIsValid} style={styles.btn(!stepIsValid)}>
            Next
          </button>
        ) : (
          <button type="submit" disabled={!stepIsValid} style={styles.btn(!stepIsValid)}>
            Submit
          </button>
        )}
      </div>
    </form>
  );
}
