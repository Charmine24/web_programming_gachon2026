// Week 13 — Topic 03
import { useState } from 'react';

// ─── 4a. Checkbox — boolean state ───────────────────────────────────────────
// Use `checked` (not `value`) and read e.target.checked.
function TermsCheckbox() {
  const [agreedToTerms, setAgreedToTerms] = useState(false);

  return (
    <label>
      <input
        type="checkbox"
        checked={agreedToTerms}
        onChange={e => setAgreedToTerms(e.target.checked)}
      />
      I agree to the Terms and Conditions
    </label>
  );
}

// ─── 4b. Multiple checkboxes — array state ──────────────────────────────────
function SkillCheckboxes() {
  const skills = ['HTML', 'CSS', 'JavaScript', 'React'];
  const [selected, setSelected] = useState([]);

  function handleSkill(skill, checked) {
    if (checked) {
      setSelected(prev => [...prev, skill]);
    } else {
      setSelected(prev => prev.filter(s => s !== skill));
    }
  }

  return (
    <div>
      {skills.map(skill => (
        <label key={skill}>
          <input
            type="checkbox"
            checked={selected.includes(skill)}
            onChange={e => handleSkill(skill, e.target.checked)}
          />
          {skill}
        </label>
      ))}
      <p>Selected: {selected.join(', ')}</p>
    </div>
  );
}

// ─── 4c. Radio buttons — string state ───────────────────────────────────────
// The same `name` groups them; checked={year === y} marks the chosen one.
function YearRadios() {
  const [year, setYear] = useState('1');

  return (
    <div>
      {['1', '2', '3', '4'].map(y => (
        <label key={y}>
          <input
            type="radio"
            name="year"
            value={y}
            checked={year === y}
            onChange={e => setYear(e.target.value)}
          />
          Year {y}
        </label>
      ))}
    </div>
  );
}

export { TermsCheckbox, SkillCheckboxes, YearRadios };

export default function Demo() {
  return (
    <div>
      <TermsCheckbox />
      <SkillCheckboxes />
      <YearRadios />
    </div>
  );
}
