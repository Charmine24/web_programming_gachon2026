// Week 13 — Homework: Practice Challenge — Course Feedback Form
// Fields: name (required), 8-digit student ID (required), star rating (required),
// topics (optional checkboxes), comments (max 300 chars, live count).
// Bonus: submit is disabled while any required field is empty.
import { useState } from 'react';

const RATINGS = [
  { value: '5', label: '⭐⭐⭐⭐⭐' },
  { value: '4', label: '⭐⭐⭐⭐' },
  { value: '3', label: '⭐⭐⭐' },
];
const TOPICS = ['HTML', 'CSS', 'JavaScript', 'Git', 'React'];
const MAX_COMMENT = 300;

const INITIAL_FORM = {
  name: '',
  studentId: '',
  rating: '',
  topics: [],
  comments: '',
};

function CourseFeedbackForm() {
  const [form, setForm] = useState(INITIAL_FORM);
  const [errors, setErrors] = useState({});
  const [submittedName, setSubmittedName] = useState('');

  // One handler for text, number, radio and textarea inputs
  function handleChange(e) {
    const { name, value } = e.target;
    const newValue = name === 'comments' ? value.slice(0, MAX_COMMENT) : value;
    setForm(prev => ({ ...prev, [name]: newValue }));
    if (errors[name]) setErrors(prev => ({ ...prev, [name]: '' }));
  }

  // Checkboxes: add or remove the topic from the array
  function handleTopic(topic, checked) {
    setForm(prev => ({
      ...prev,
      topics: checked
        ? [...prev.topics, topic]
        : prev.topics.filter(t => t !== topic),
    }));
  }

  function validate() {
    const errs = {};
    if (!form.name.trim()) {
      errs.name = 'Student name is required';
    }
    if (!form.studentId) {
      errs.studentId = 'Student ID is required';
    } else if (!/^\d{8}$/.test(form.studentId)) {
      errs.studentId = 'Student ID must be exactly 8 digits';
    }
    if (!form.rating) {
      errs.rating = 'Please choose a rating';
    }
    return errs;
  }

  function handleSubmit(e) {
    e.preventDefault(); // stop the page reloading
    const errs = validate();
    if (Object.keys(errs).length > 0) {
      setErrors(errs);
      return;
    }
    console.log('Feedback submitted:', form);
    setSubmittedName(form.name.trim());
    setForm(INITIAL_FORM);
    setErrors({});
  }

  // Bonus: disable submit while any required field is empty
  const requiredEmpty =
    !form.name.trim() || !form.studentId || !form.rating;

  if (submittedName) {
    return (
      <div>
        <h2>Thank you, {submittedName}!</h2>
        <p>Your feedback has been received.</p>
        <button onClick={() => setSubmittedName('')}>Submit another response</button>
      </div>
    );
  }

  const errorStyle = { color: 'red', fontSize: 13, margin: '4px 0' };
  const remaining = MAX_COMMENT - form.comments.length;

  return (
    <form onSubmit={handleSubmit} noValidate style={{ maxWidth: 420 }}>
      <h2>Course Feedback</h2>

      {/* 1. Student Name */}
      <div>
        <label>
          Student Name *<br />
          <input
            type="text"
            name="name"
            value={form.name}
            onChange={handleChange}
            style={{ borderColor: errors.name ? 'red' : '' }}
          />
        </label>
        {errors.name && <p style={errorStyle}>{errors.name}</p>}
      </div>

      {/* 2. Student ID */}
      <div>
        <label>
          Student ID (8 digits) *<br />
          <input
            type="number"
            name="studentId"
            value={form.studentId}
            onChange={handleChange}
            style={{ borderColor: errors.studentId ? 'red' : '' }}
          />
        </label>
        {errors.studentId && <p style={errorStyle}>{errors.studentId}</p>}
      </div>

      {/* 3. Rating */}
      <fieldset>
        <legend>Rating *</legend>
        {RATINGS.map(r => (
          <label key={r.value} style={{ display: 'block' }}>
            <input
              type="radio"
              name="rating"
              value={r.value}
              checked={form.rating === r.value}
              onChange={handleChange}
            />
            {r.label}
          </label>
        ))}
        {errors.rating && <p style={errorStyle}>{errors.rating}</p>}
      </fieldset>

      {/* 4. Topics Covered Well */}
      <fieldset>
        <legend>Topics Covered Well (optional)</legend>
        {TOPICS.map(topic => (
          <label key={topic} style={{ marginRight: 12 }}>
            <input
              type="checkbox"
              checked={form.topics.includes(topic)}
              onChange={e => handleTopic(topic, e.target.checked)}
            />
            {topic}
          </label>
        ))}
      </fieldset>

      {/* 5. Comments */}
      <div>
        <label>
          Comments<br />
          <textarea
            name="comments"
            value={form.comments}
            onChange={handleChange}
            rows={4}
            cols={40}
          />
        </label>
        <p style={{ color: remaining < 30 ? 'red' : 'gray', fontSize: 13, margin: '4px 0' }}>
          {form.comments.length} / {MAX_COMMENT}
        </p>
      </div>

      {/* 6. Submit */}
      <button type="submit" disabled={requiredEmpty}>
        Submit Feedback
      </button>
    </form>
  );
}

export default CourseFeedbackForm;
