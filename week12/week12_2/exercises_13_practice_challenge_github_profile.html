Practice Challenge: GitHubProfile

A GitHubProfile component that fetches a GitHub user as you type, debounced by 500ms so it does not fire on every keystroke.

Requirements

  1. An <input> where the user types a GitHub username ✔
  2. When the username changes, fetch https://api.github.com/users/{username} ✔ effect with [username]
  3. Show a loading indicator while fetching ✔
  4. Display avatar, name, public_repos count and bio ✔
  5. Show an error message if the user is not found (404) ✔
  6. Bonus: debounce the fetch by 500ms ✔ setTimeout + clearTimeout in cleanup (plus AbortController for stale requests)

React version (my solution)

import { useState, useEffect } from 'react';

export default function GitHubProfile() {
  const [username, setUsername] = useState('muxiddin19');
  const [user, setUser]         = useState(null);
  const [loading, setLoading]   = useState(false);
  const [error, setError]       = useState(null);

  useEffect(() => {
    const name = username.trim();
    if (!name) { setUser(null); setError(null); setLoading(false); return; }

    const controller = new AbortController();
    setLoading(true);
    setError(null);

    // Debounce: wait 500ms after the last keystroke before fetching
    const timerId = setTimeout(async () => {
      try {
        const res = await fetch(`https://api.github.com/users/${name}`,
                                { signal: controller.signal });
        if (res.status === 404) throw new Error(`User "${name}" not found (404)`);
        if (!res.ok) throw new Error('Request failed: ' + res.status);
        const data = await res.json();
        setUser(data);
      } catch (err) {
        if (err.name === 'AbortError') return;   // a newer request replaced this one
        setUser(null);
        setError(err.message);
      } finally {
        if (!controller.signal.aborted) setLoading(false);
      }
    }, 500);

    // Cleanup: cancel the timer AND any request still in flight
    return () => {
      clearTimeout(timerId);
      controller.abort();
    };
  }, [username]);

  return (
    <div>
      <input value={username} placeholder="GitHub username"
             onChange={e => setUsername(e.target.value)} />

      {loading && <p>Loading...</p>}
      {error && <p style={{ color: 'crimson' }}>{error}</p>}
      {user && !loading && (
        <div className="profile">
          <img src={user.avatar_url} alt={`${user.login}'s avatar`} width="96" height="96" />
          <div>
            <h3>{user.name || user.login}</h3>
            <p>@{user.login}</p>
            <span className="repos">{user.public_repos} public repos</span>
            <p>{user.bio || 'No bio yet.'}</p>
          </div>
        </div>
      )}
    </div>
  );
}
